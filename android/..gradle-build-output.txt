// Configuration
const DELAY_BETWEEN_REQUESTS = 2000; // 2 seconds to be safe
const MAX_RETRIES = 3;
const BATCH_SIZE = 50; // Process in batches to save progress

/**
 * Sleep function for rate limiting
 */
function sleep(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

/**
 * Parse product name to improve search results
 */
function cleanProductName(productName) {
  // Keep the full product name for better search results
  return productName.trim();
}
/**
 * Search Google for product images using your custom search engine
 */
async function searchGoogleImage(productName, retryCount = 0) {
  try {
    // Clean the product name for better search
    const cleanName = cleanProductName(productName);
    const searchQuery = `${cleanName} product`;

    console.log(`🔍 Searching: "${searchQuery}"`);

    const response = await axios.get(
      "https://www.googleapis.com/customsearch/v1",
      {
        params: {
          key: GOOGLE_API_KEY,
          cx: GOOGLE_CX,
          q: searchQuery,
          searchType: "image",
          num: 5, // Get top 5 results
          imgSize: "large",
          imgType: "photo",
          safe: "active",
          imgDominantColor: "white", // Prefer white backgrounds
        },
        timeout: 10000,
      }
    );

    if (response.data.items && response.data.items.length > 0) {
      // Filter and sort images by quality
      const validImages = response.data.items
        .filter((item) => {
          const url = item.link.toLowerCase();
          // Avoid SVG and prefer JPG/PNG
          return (
            !url.includes("svg") &&
            (url.includes("jpg") ||
              url.includes("jpeg") ||
              url.includes("png") ||
              url.includes("webp"))
          );
        })
        .sort((a, b) => {
          // Prefer larger images
          const sizeA = (a.image?.width || 0) * (a.image?.height || 0);
          const sizeB = (b.image?.width || 0) * (b.image?.height || 0);
          return sizeB - sizeA;
        });

      if (validImages.length > 0) {
        const bestImage = validImages[0];
        const dimensions = bestImage.image
          ? `${bestImage.image.width}x${bestImage.image.height}`
          : "unknown";

        console.log(
          `✅ Found image (${dimensions}): ${bestImage.link.substring(
            0,
            50
          )}...`
        );
        return bestImage.link;
      }
    }

    console.log(`⚠️  No suitable images found for: ${productName}`);
    return null;
  } catch (error) {
    // Handle rate limiting
    if (error.response?.status === 429 && retryCount < MAX_RETRIES) {
      const waitTime = (retryCount + 1) * 5000; // Progressive backoff
      console.log(
        `⏳ Rate limit hit, waiting ${waitTime / 1000}s before retry...`
      );
      await sleep(waitTime);
      return searchGoogleImage(productName, retryCount + 1);
    }

    // Handle quota exceeded
    if (error.response?.status === 429) {
      console.error(
        `❌ Daily quota exceeded. You've used your 100 free queries for today.`
      );
      console.error(`   Please wait 24 hours or upgrade to paid tier.`);
      throw new Error("QUOTA_EXCEEDED");
    }

    console.error(`❌ Search error for "${productName}":`, error.message);
    if (error.response?.data?.error) {
      console.error(`   Details:`, error.response.data.error.message);
    }
    return null;
  }
}

/**
 * Validate image URL before uploading
 */
async function validateImage(imageUrl) {
  try {
    const response = await axios.head(imageUrl, {
      timeout: 5000,
      maxRedirects: 5,
    });

    const contentType = response.headers["content-type"];
    const contentLength = parseInt(response.headers["content-length"] || 0);

    // Check if it's an image and reasonable size
    const isImage = contentType && contentType.startsWith("image/");
    const isReasonableSize = contentLength > 1000 && contentLength < 20000000; // 1KB - 20MB

    if (!isImage) {
      console.log(`⚠️  Not a valid image (content-type: ${contentType})`);
      return false;
    }

    if (!isReasonableSize) {
      console.log(`⚠️  Image size out of range: ${contentLength} bytes`);
      return false;
    }

    return true;
  } catch (error) {
    console.log(`⚠️  Could not validate image: ${error.message}`);
    return false;
  }
}

/**
 * Upload image to Cloudinary
 */
async function uploadToCloudinary(imageUrl, productName, productId) {
  try {
    // Validate image first
    console.log(`🔍 Validating image...`);
    const isValid = await validateImage(imageUrl);

    if (!isValid) {
      console.log(`❌ Image validation failed, skipping upload`);
      return null;
    }

    // Create a safe filename
    const safeFilename = productName
      .replace(/[^a-zA-Z0-9]/g, "_")
      .toLowerCase()
      .substring(0, 80);

    console.log(`📤 Uploading to Cloudinary...`);

    const result = await cloudinary.uploader.upload(imageUrl, {
      folder: "terango/products",
      public_id: `${safeFilename}_${productId.substring(0, 8)}`,
      overwrite: false,
      resource_type: "image",
      transformation: [
        { width: 1000, height: 1000, crop: "limit" }, // Max 1000x1000
        { quality: "auto:good" }, // Auto optimize quality
        { fetch_format: "auto" }, // Auto format (WebP when supported)
      ],
    });

    console.log(`✅ Uploaded successfully!`);
    console.log(`   URL: ${result.secure_url}`);
    return result.secure_url;
  } catch (error) {
    console.error(`❌ Cloudinary upload failed:`, error.message);
    return null;
  }
}

/**
 * Save progress to JSON file
 */
function saveProgress(data) {
  const fs = require("fs");
  fs.writeFileSync("image-upload-progress.json", JSON.stringify(data, null, 2));
}

/**
 * Load previous progress
 */
function loadProgress() {
  try {
    const fs = require("fs");
    if (fs.existsSync("image-upload-progress.json")) {
      const data = fs.readFileSync("image-upload-progress.json", "utf8");
      return JSON.parse(data);
    }
  } catch (error) {
    console.log("No previous progress found, starting fresh.");
  }
  return { processedIds: [], lastIndex: 0 };
}

/**
 * Main function to process all products without images
 */
async function processProductImages() {
  console.log("🚀 TerranGO Product Image Automation");
  console.log("=".repeat(60));
  console.log(`📡 Search Engine ID: ${GOOGLE_CX}`);
  console.log(`📊 Daily Limit: 100 queries (Google Free Tier)`);
  console.log("=".repeat(60));
  console.log("");

  try {
    // Load previous progress
    const progress = loadProgress();
    console.log(
      `📂 Loaded progress: ${progress.processedIds.length} products already processed\n`
    );

    // Get all products without images
    const productsWithoutImages = await prisma.product.findMany({
      where: {
        shopId: SHOP_ID,
        OR: [{ imageUrl: null }, { imageUrl: "" }],
        id: {
          notIn: progress.processedIds, // Skip already processed
        },
      },
      select: {
        id: true,
        name: true,
      },
      orderBy: {
        name: "asc",
      },
    });

    const totalProducts = productsWithoutImages.length;
    console.log(`📊 Found ${totalProducts} products without images\n`);

    if (totalProducts === 0) {
      console.log("✅ All products already have images!");
      return;
    }

    // Warning about daily limits
    if (totalProducts > 90) {
      console.log("⚠️  WARNING: You have more than 90 products to process.");
      console.log("   Google's free tier allows 100 queries/day.");
      console.log("   The script will stop if quota is exceeded.\n");
    }

    let successCount = 0;
    let failCount = 0;
    let skippedCount = 0;

    // Process each product
    for (let i = 0; i < totalProducts; i++) {
      const product = productsWithoutImages[i];
      const progressPercent = (((i + 1) / totalProducts) * 100).toFixed(1);

      console.log(`\n${"=".repeat(60)}`);
      console.log(
        `[${i + 1}/${totalProducts}] ${progressPercent}% - ${product.name}`
      );
      console.log("─".repeat(60));

      try {
        // Step 1: Search for image
        const imageUrl = await searchGoogleImage(product.name);

        if (!imageUrl) {
          console.log(`⏭️  No image found, skipping...`);
          skippedCount++;
          progress.processedIds.push(product.id);
          saveProgress(progress);
          await sleep(DELAY_BETWEEN_REQUESTS);
          continue;
        }

        // Step 2: Upload to Cloudinary
        const cloudinaryUrl = await uploadToCloudinary(
          imageUrl,
          product.name,
          product.id
        );

        if (!cloudinaryUrl) {
          console.log(`⏭️  Upload failed, skipping...`);
          failCount++;
          progress.processedIds.push(product.id);
          saveProgress(progress);
          await sleep(DELAY_BETWEEN_REQUESTS);
          continue;
        }

        // Step 3: Update database
        await prisma.product.update({
          where: { id: product.id },
          data: { imageUrl: cloudinaryUrl },
        });

        successCount++;
        progress.processedIds.push(product.id);
        console.log(`🎉 SUCCESS! Database updated.`);

        // Save progress every 10 products
        if ((i + 1) % 10 === 0) {
          saveProgress(progress);
          console.log(
            `\n💾 Progress saved (${successCount} successful so far)`
          );
        }
      } catch (error) {
        if (error.message === "QUOTA_EXCEEDED") {
          console.log(`\n⛔ Stopping due to quota limit.`);
          console.log(`   Processed: ${i + 1}/${totalProducts}`);
          console.log(`   Resume tomorrow by running the script again.`);
          saveProgress(progress);
          break;
        }
        throw error;
      }

      // Rate limiting delay
      await sleep(DELAY_BETWEEN_REQUESTS);
    }

    // Final save
    saveProgress(progress);

    // Summary
    console.log("\n" + "=".repeat(60));
    console.log("📊 FINAL SUMMARY");
    console.log("=".repeat(60));
    console.log(`✅ Successful uploads: ${successCount}`);
    console.log(`❌ Failed uploads: ${failCount}`);
    console.log(`⏭️  No image found: ${skippedCount}`);
    console.log(
      `📦 Total processed: ${successCount + failCount + skippedCount}`
    );
    console.log(
      `📋 Remaining: ${
        totalProducts - (successCount + failCount + skippedCount)
      }`
    );
    console.log("=".repeat(60));

    if (successCount > 0) {
      console.log(`\n🎉 Successfully uploaded ${successCount} product images!`);
    }

    if (totalProducts > successCount + failCount + skippedCount) {
      console.log(
        `\n💡 TIP: Run this script again tomorrow to process remaining products.`
      );
    }
  } catch (error) {
    console.error("\n💥 Fatal error:", error);
    throw error;
  } finally {
    await prisma.$disconnect();
  }
}

// Run the script
processProductImages()
